cmake_minimum_required(VERSION 3.16)

target_compile_features(NonconvexSpinMotion PRIVATE cxx_std_17)
target_link_libraries(NonconvexSpinMotion PRIVATE gz-plugin2::gz-plugin2 gz-sim8::gz-sim8)
