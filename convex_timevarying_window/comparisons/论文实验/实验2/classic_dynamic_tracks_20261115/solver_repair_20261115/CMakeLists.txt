cmake_minimum_required(VERSION 3.16)
project(classic_solver_repair LANGUAGES CXX)
get_filename_component(RUNTIME "${CMAKE_CURRENT_SOURCE_DIR}/../../cpp_runtime" ABSOLUTE)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
add_subdirectory("${RUNTIME}" runtime)

add_executable(repair_continuation_cached_sip sip_continuation_cached.cpp "${REPO}/静态凸窗口/复现/experiments/free_velocity/minco_pv.cpp")
target_include_directories(repair_continuation_cached_sip PRIVATE "${RUNTIME}/include" "${CASADI_ROOT}/include" "${CASADI_ROOT}/include/eigen3" "${REPO}/静态凸窗口/复现/experiments/free_velocity" "${REPO}/静态凸窗口/复现/TOGT-Planner-ICRA2024/include")
target_link_libraries(repair_continuation_cached_sip PRIVATE "${CASADI_ROOT}/libcasadi.so" "${CASADI_ROOT}/../numpy.libs/libopenblas64_p-r0-0cf96a72.3.23.dev.so")
target_compile_features(repair_continuation_cached_sip PRIVATE cxx_std_17)
target_compile_options(repair_continuation_cached_sip PRIVATE -O3 -ffp-contract=off)
set_target_properties(repair_continuation_cached_sip PROPERTIES BUILD_RPATH "${CASADI_ROOT};${CASADI_ROOT}/../numpy.libs")
target_link_options(repair_continuation_cached_sip PRIVATE -Wl,--disable-new-dtags)

foreach(name homotopy_prepare_cached homotopy_run homotopy_adaptive homotopy_static_inputs)
  add_executable(${name} ${name}.cpp)
  target_include_directories(${name} PRIVATE "${RUNTIME}/include" "${CASADI_ROOT}/include/eigen3" "${REPO}/静态凸窗口/复现/TOGT-Planner-ICRA2024/include")
  target_compile_features(${name} PRIVATE cxx_std_17)
  target_compile_options(${name} PRIVATE -O3 -ffp-contract=off)
  target_link_libraries(${name} PRIVATE dynamic_togt_upstream)
endforeach()

add_executable(repair_lift lift.cpp "${REPO}/静态凸窗口/复现/experiments/free_velocity/minco_pv.cpp")
target_include_directories(repair_lift PRIVATE "${RUNTIME}/include" "${CASADI_ROOT}/include/eigen3" "${REPO}/静态凸窗口/复现/experiments/free_velocity" "${REPO}/静态凸窗口/复现/TOGT-Planner-ICRA2024/include")
target_compile_features(repair_lift PRIVATE cxx_std_17)
target_compile_options(repair_lift PRIVATE -O3 -ffp-contract=off)
