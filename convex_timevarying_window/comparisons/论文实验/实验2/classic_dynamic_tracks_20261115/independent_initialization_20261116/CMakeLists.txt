cmake_minimum_required(VERSION 3.16)
project(independent_initial LANGUAGES CXX)
get_filename_component(RUNTIME "${CMAKE_CURRENT_SOURCE_DIR}/../../cpp_runtime" ABSOLUTE)
get_filename_component(REPO "${RUNTIME}/../../../../.." ABSOLUTE)
add_subdirectory("${RUNTIME}" runtime)
add_executable(independent_prepare prepare.cpp "${REPO}/静态凸窗口/复现/experiments/free_velocity/minco_pv.cpp")
target_include_directories(independent_prepare PRIVATE "${RUNTIME}/include" "${CASADI_ROOT}/include" "${REPO}/静态凸窗口/复现/experiments/free_velocity")
target_link_libraries(independent_prepare PRIVATE dynamic_togt_upstream "${CASADI_ROOT}/libcasadi.so")
target_compile_features(independent_prepare PRIVATE cxx_std_17)
target_compile_options(independent_prepare PRIVATE -O3 -ffp-contract=off)
set_target_properties(independent_prepare PROPERTIES BUILD_RPATH "${CASADI_ROOT}")

add_executable(independent_derivative_check derivative_check.cpp "${REPO}/静态凸窗口/复现/experiments/free_velocity/minco_pv.cpp")
target_include_directories(independent_derivative_check PRIVATE "${RUNTIME}/include" "${CASADI_ROOT}/include" "${CASADI_ROOT}/include/eigen3" "${REPO}/静态凸窗口/复现/experiments/free_velocity" "${REPO}/静态凸窗口/复现/TOGT-Planner-ICRA2024/include")
target_link_libraries(independent_derivative_check PRIVATE "${CASADI_ROOT}/libcasadi.so")
target_compile_features(independent_derivative_check PRIVATE cxx_std_17)
target_compile_options(independent_derivative_check PRIVATE -O2)
set_target_properties(independent_derivative_check PROPERTIES BUILD_RPATH "${CASADI_ROOT}")
