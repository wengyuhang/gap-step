cmake_minimum_required(VERSION 3.16)
project(compact_slalom_static_control LANGUAGES CXX)
get_filename_component(RUNTIME "${CMAKE_CURRENT_SOURCE_DIR}/../../../cpp_runtime" ABSOLUTE)
get_filename_component(REPO "${RUNTIME}/../../../../.." ABSOLUTE)
add_subdirectory("${RUNTIME}" runtime)
add_executable(static_control_inputs generate.cpp "${REPO}/静态凸窗口/复现/experiments/free_velocity/minco_pv.cpp")
target_include_directories(static_control_inputs PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../.." "${RUNTIME}/include" "${CASADI_ROOT}/include" "${REPO}/静态凸窗口/复现/experiments/free_velocity")
target_link_libraries(static_control_inputs PRIVATE dynamic_togt_upstream "${CASADI_ROOT}/libcasadi.so")
target_compile_features(static_control_inputs PRIVATE cxx_std_17)
target_compile_options(static_control_inputs PRIVATE -O3 -ffp-contract=off)
set_target_properties(static_control_inputs PROPERTIES BUILD_RPATH "${CASADI_ROOT}")
