cmake_minimum_required(VERSION 3.16)
project(compact_slalom_scene LANGUAGES CXX)
get_filename_component(RUNTIME "${CMAKE_CURRENT_SOURCE_DIR}/../../cpp_runtime" ABSOLUTE)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
add_subdirectory("${RUNTIME}" runtime)
add_executable(slalom_inputs generate.cpp "${REPO}/静态凸窗口/复现/experiments/free_velocity/minco_pv.cpp")
target_include_directories(slalom_inputs PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/.." "${RUNTIME}/include" "${CASADI_ROOT}/include" "${REPO}/静态凸窗口/复现/experiments/free_velocity")
target_link_libraries(slalom_inputs PRIVATE dynamic_togt_upstream "${CASADI_ROOT}/libcasadi.so")
target_compile_features(slalom_inputs PRIVATE cxx_std_17)
target_compile_options(slalom_inputs PRIVATE -O3 -ffp-contract=off)
set_target_properties(slalom_inputs PROPERTIES BUILD_RPATH "${CASADI_ROOT}")
