cmake_minimum_required(VERSION 3.16)
project(closed_course LANGUAGES CXX)
get_filename_component(RUNTIME "${CMAKE_CURRENT_SOURCE_DIR}/../../../cpp_runtime" ABSOLUTE)
get_filename_component(REPO "${RUNTIME}/../../../../.." ABSOLUTE)
add_subdirectory("${RUNTIME}" runtime)
add_executable(long_closed_inputs generate.cpp "${REPO}/静态凸窗口/复现/experiments/free_velocity/minco_pv.cpp")
target_include_directories(long_closed_inputs PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../.." "${RUNTIME}/include" "${CASADI_ROOT}/include" "${REPO}/静态凸窗口/复现/experiments/free_velocity")

target_compile_options(long_closed_inputs PRIVATE -O3 -ffp-contract=off)
set_target_properties(long_closed_inputs PROPERTIES BUILD_RPATH "${CASADI_ROOT}")
