cmake_minimum_required(VERSION 3.18)
project(slsqp_native_minco LANGUAGES CXX Fortran)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CASADI_ROOT "" CACHE PATH "CasADi C++ SDK")
if(NOT EXISTS "${CASADI_ROOT}/include/casadi/casadi.hpp")
  message(FATAL_ERROR "Set CASADI_ROOT")
endif()
set(NUMPY_BLAS "${CASADI_ROOT}/../numpy.libs/libopenblas64_p-r0-0cf96a72.3.23.dev.so" CACHE FILEPATH "Original ILP64 OpenBLAS")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/cases")
file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/assets/cases.tar.gz" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/cases")
add_library(slsqp_core STATIC vendor/slsqp/slsqp_optmz.f)
target_compile_options(slsqp_core PRIVATE -O3 -std=legacy -frecursive)
add_library(native_minco_pv STATIC vendor/minco_pv/minco_pv.cpp)
target_include_directories(native_minco_pv PUBLIC vendor/minco_pv vendor/togt/include "${CASADI_ROOT}/include/eigen3")
target_compile_options(native_minco_pv PRIVATE -O3 -ffp-contract=off)
add_executable(sip_planner src/sip_planner.cpp)
add_executable(matrix src/matrix.cpp)
add_executable(mesh_check src/mesh_check.cpp)
add_executable(hard_unit tests/hard_unit.cpp)
foreach(target sip_planner matrix mesh_check hard_unit)
  target_include_directories(${target} PRIVATE include "${CASADI_ROOT}/include" "${CASADI_ROOT}/include/eigen3")
  target_compile_options(${target} PRIVATE -O3 -ffp-contract=off)
endforeach()
foreach(target sip_planner hard_unit)
  target_link_libraries(${target} PRIVATE native_minco_pv slsqp_core "${CASADI_ROOT}/libcasadi.so" "${NUMPY_BLAS}")
  set_target_properties(${target} PROPERTIES BUILD_RPATH "${CASADI_ROOT};${CASADI_ROOT}/../numpy.libs")
  target_link_options(${target} PRIVATE -Wl,--disable-new-dtags)
endforeach()
enable_testing()
foreach(method ours center)
  add_test(NAME hard_${method}_gradient COMMAND hard_unit "${CMAKE_CURRENT_BINARY_DIR}/cases/inputs/a25_f6_${method}"
    "${CMAKE_CURRENT_BINARY_DIR}/cases/assets/a25_f6/scene.json" "${CMAKE_CURRENT_BINARY_DIR}/hard_${method}_validation.json")
  set_tests_properties(hard_${method}_gradient PROPERTIES ENVIRONMENT "OPENBLAS_NUM_THREADS=1;OMP_NUM_THREADS=1")
endforeach()
add_executable(experiment src/experiment.cpp)
target_include_directories(experiment PRIVATE include)
target_compile_options(experiment PRIVATE -O3)
