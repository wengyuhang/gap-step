cmake_minimum_required(VERSION 3.16)
project(window_racing_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CASADI_ROOT "" CACHE PATH "Installed CasADi SDK root")
if(NOT EXISTS "${CASADI_ROOT}/include/casadi/casadi.hpp")
  message(FATAL_ERROR "Set CASADI_ROOT to the installed CasADi directory")
endif()
add_executable(window_native src/graph_runner.cpp)
target_compile_options(window_native PRIVATE -O3)
target_compile_definitions(window_native PRIVATE _GLIBCXX_USE_CXX11_ABI=1)
target_include_directories(window_native PRIVATE "${CASADI_ROOT}/include")
target_link_libraries(window_native PRIVATE "${CASADI_ROOT}/libcasadi.so")
set_target_properties(window_native PROPERTIES BUILD_RPATH "${CASADI_ROOT}")
add_executable(frame_check src/frame_check.cpp)
target_include_directories(frame_check PRIVATE "${CASADI_ROOT}/include/eigen3" include)
target_compile_options(frame_check PRIVATE -O3)
add_executable(sip_planner src/sip_planner.cpp)
target_include_directories(sip_planner PRIVATE include "${CASADI_ROOT}/include" "${CASADI_ROOT}/include/eigen3")
target_compile_options(sip_planner PRIVATE -O3)
target_link_libraries(sip_planner PRIVATE "${CASADI_ROOT}/libcasadi.so")
set_target_properties(sip_planner PROPERTIES BUILD_RPATH "${CASADI_ROOT}")
enable_language(C)
add_library(original_lbfgsb STATIC vendor/scipy/lbfgsb.c)
file(GLOB SCIPY_BLAS "${CASADI_ROOT}/../scipy.libs/libscipy_openblas*.so")
target_link_libraries(original_lbfgsb PUBLIC ${SCIPY_BLAS} m)
add_executable(togt_planner src/togt_planner.cpp)
target_include_directories(togt_planner PRIVATE include "${CASADI_ROOT}/include" "${CASADI_ROOT}/include/eigen3" "${CMAKE_CURRENT_SOURCE_DIR}/../soft_dynamics/vendor/include")
target_compile_options(togt_planner PRIVATE -O3)
target_link_libraries(togt_planner PRIVATE "${CASADI_ROOT}/libcasadi.so" original_lbfgsb)
set_target_properties(togt_planner PROPERTIES BUILD_RPATH "${CASADI_ROOT}")
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
add_executable(wang_planner src/wang_planner.cpp "${REPO_ROOT}/复现/wang2023_verified/native.cpp")
target_include_directories(wang_planner PRIVATE include "${CASADI_ROOT}/include" "${CASADI_ROOT}/include/eigen3" "${REPO_ROOT}/复现/wang2023_verified/vendor/LBFGS-Lite/include")
target_compile_options(wang_planner PRIVATE -O3 -DNDEBUG)
target_link_libraries(wang_planner PRIVATE "${CASADI_ROOT}/libcasadi.so")
set_target_properties(wang_planner PROPERTIES BUILD_RPATH "${CASADI_ROOT}")

add_executable(mesh_check src/mesh_check.cpp)
target_include_directories(mesh_check PRIVATE include "${CASADI_ROOT}/include/eigen3")
target_compile_options(mesh_check PRIVATE -O3)
add_executable(matrix src/matrix.cpp)
target_include_directories(matrix PRIVATE include "${CASADI_ROOT}/include/eigen3")
target_compile_options(matrix PRIVATE -O3)
add_executable(compile_wang src/compile_wang.cpp)
target_include_directories(compile_wang PRIVATE include "${CASADI_ROOT}/include")
target_link_libraries(compile_wang PRIVATE "${CASADI_ROOT}/libcasadi.so")
set_target_properties(compile_wang PROPERTIES BUILD_RPATH "${CASADI_ROOT}")
add_executable(wang_batch src/wang_batch.cpp)
target_include_directories(wang_batch PRIVATE include "${CASADI_ROOT}/include/eigen3")
target_compile_options(wang_batch PRIVATE -O3)
enable_testing()
add_executable(native_unit tests/native_unit.cpp)
target_include_directories(native_unit PRIVATE include "${CASADI_ROOT}/include/eigen3")
target_compile_options(native_unit PRIVATE -O3)
target_link_libraries(native_unit PRIVATE original_lbfgsb)
add_test(NAME native_unit COMMAND native_unit)
add_executable(sip_probe tests/sip_probe.cpp)
target_include_directories(sip_probe PRIVATE include "${CASADI_ROOT}/include" "${CASADI_ROOT}/include/eigen3")
target_compile_options(sip_probe PRIVATE -O3)
target_link_libraries(sip_probe PRIVATE "${CASADI_ROOT}/libcasadi.so")
set_target_properties(sip_probe PROPERTIES BUILD_RPATH "${CASADI_ROOT}")
add_executable(native_report src/report.cpp)
target_include_directories(native_report PRIVATE include "${CASADI_ROOT}/include/eigen3")
target_compile_options(native_report PRIVATE -O3)

# Match the numerical kernels of the original wyh NumPy geometry oracle.
set(NUMPY_BLAS "${CASADI_ROOT}/../numpy.libs/libopenblas64_p-r0-0cf96a72.3.23.dev.so"
    CACHE FILEPATH "Original NumPy ILP64 OpenBLAS library")
if(NOT EXISTS "${NUMPY_BLAS}")
  message(FATAL_ERROR "The original NumPy ILP64 OpenBLAS library is required for SIP parity")
endif()
add_executable(numpy_frame_check tests/numpy_frame_check.cpp)
target_include_directories(numpy_frame_check PRIVATE include "${CASADI_ROOT}/include/eigen3")
target_compile_options(numpy_frame_check PRIVATE -O3)
foreach(target sip_planner sip_probe numpy_frame_check)
  target_link_libraries(${target} PRIVATE "${NUMPY_BLAS}")
  target_compile_options(${target} PRIVATE -ffp-contract=off)
  set_target_properties(${target} PROPERTIES BUILD_RPATH "${CASADI_ROOT};${CASADI_ROOT}/../numpy.libs")
  target_link_options(${target} PRIVATE -Wl,--disable-new-dtags)
endforeach()
