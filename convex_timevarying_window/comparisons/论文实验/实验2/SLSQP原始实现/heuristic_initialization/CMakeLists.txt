cmake_minimum_required(VERSION 3.18)
project(window_initialization LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CASADI_ROOT "" CACHE PATH "CasADi SDK containing its Eigen headers")
if(NOT EXISTS "${CASADI_ROOT}/include/casadi/casadi.hpp")
  message(FATAL_ERROR "Set CASADI_ROOT to the current wyh CasADi SDK")
endif()
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/libcmaes_source")
file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/libcmaes.tar.gz"
  DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/libcmaes_source")
set(LIBCMAES_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LIBCMAES_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LIBCMAES_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(LIBCMAES_ENABLE_SURROG OFF CACHE BOOL "" FORCE)
set(LIBCMAES_USE_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/libcmaes_source" "${CMAKE_CURRENT_BINARY_DIR}/libcmaes_build")
set(NATIVE_PACKAGE "${CMAKE_CURRENT_SOURCE_DIR}/../../SLSQP的c++实现")
add_executable(generate_initials src/generate_initials.cpp)
target_include_directories(generate_initials PRIVATE
  "${CASADI_ROOT}/include" "${CASADI_ROOT}/include/eigen3"
  "${NATIVE_PACKAGE}/include" "${NATIVE_PACKAGE}/vendor/togt/include")
target_link_libraries(generate_initials PRIVATE libcmaes::cmaes "${CASADI_ROOT}/libcasadi.so")
target_compile_options(generate_initials PRIVATE -O3 -ffp-contract=off)
set_target_properties(generate_initials PROPERTIES BUILD_RPATH "${CASADI_ROOT};${CMAKE_CURRENT_BINARY_DIR}/libcmaes_build/src")
enable_testing()
add_test(NAME cma_rotated_quadratic COMMAND generate_initials --unit)
