cmake_minimum_required(VERSION 3.18)
project(full_state_de LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CASADI_ROOT "" CACHE PATH "CasADi SDK")
find_package(OpenMP REQUIRED)
set(NATIVE_PACKAGE "${CMAKE_CURRENT_SOURCE_DIR}/../../SLSQP的c++实现")
add_executable(generate_de src/generate_de.cpp "${NATIVE_PACKAGE}/vendor/minco_pv/minco_pv.cpp")
target_include_directories(generate_de PRIVATE "${CASADI_ROOT}/include" "${CASADI_ROOT}/include/eigen3" "${NATIVE_PACKAGE}/include" "${NATIVE_PACKAGE}/vendor/togt/include" "${NATIVE_PACKAGE}/vendor/minco_pv")
target_link_libraries(generate_de PRIVATE "${CASADI_ROOT}/libcasadi.so" OpenMP::OpenMP_CXX)
target_compile_options(generate_de PRIVATE -O3 -ffp-contract=off)
set_target_properties(generate_de PROPERTIES BUILD_RPATH "${CASADI_ROOT}")
